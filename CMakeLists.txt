  if(test STREQUAL "pt_tests" OR test MATCHES "^pt_.*_test$")
    if(MSVC)
      target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
  endif()
endforeach()
