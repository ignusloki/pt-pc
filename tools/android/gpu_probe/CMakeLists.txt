cmake_minimum_required(VERSION 3.22)
project(pt_android_gpu_probe LANGUAGES CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Configure with the Android NDK toolchain")
endif()
find_library(PT_PROBE_VULKAN vulkan REQUIRED)
add_executable(pt_android_gpu_probe main.cpp)
target_compile_features(pt_android_gpu_probe PRIVATE cxx_std_17)
target_link_libraries(pt_android_gpu_probe PRIVATE ${PT_PROBE_VULKAN})
